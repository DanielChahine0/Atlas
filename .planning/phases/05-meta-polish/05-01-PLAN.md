  type SafeMethod = (typeof SAFE_METHODS)[number];
  export function toOutboxIntent(e: WireEvent): OutboxIntent {
    let path: string; let method: SafeMethod; let headers: Record<string,string>;
    switch (e.op) {
      case "increment": { ... path=...; method="PATCH"; headers={...}; break; }
      case "upsert":    { const note=...; path=`/vault/${note}.md`; method="PATCH"; headers={Operation:"replace",...}; break; }
      case "append":    { ... method="POST"; ... break; }
      default: { const bad: never = e.op; throw new Error(...); }
    }
    if (!SAFE_METHODS.includes(method)) throw new Error(`refusing non-safe outbound method: ${method}`);
    headers["X-Atlas-Idem"] = e.idempotencyKey;
    return { idem: e.idempotencyKey, path, method, headers: JSON.stringify(headers), body: JSON.stringify(e.payload) };
  }
